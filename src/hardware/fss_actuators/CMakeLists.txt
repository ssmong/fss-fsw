cmake_minimum_required(VERSION 3.16)
project(fss_actuators)

find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(fss_interfaces REQUIRED)
find_package(fss_hal REQUIRED)
find_package(std_msgs REQUIRED)
find_package(std_srvs REQUIRED)
find_package(yaml-cpp REQUIRED)
find_package(nlohmann_json REQUIRED)

add_library(fss_valve_map src/valve_thrust_map.cpp)
target_compile_features(fss_valve_map PUBLIC cxx_std_17)
target_include_directories(fss_valve_map PUBLIC include)
target_link_libraries(fss_valve_map PUBLIC yaml-cpp)

add_executable(valve_driver_node src/valve_driver_node.cpp)
target_compile_features(valve_driver_node PRIVATE cxx_std_17)
target_compile_options(valve_driver_node PRIVATE -Wall -Wextra -Wpedantic)
target_link_libraries(valve_driver_node fss_valve_map fss_hal::fss_hal_cpp)
ament_target_dependencies(valve_driver_node rclcpp rcl_interfaces ament_index_cpp fss_interfaces)

add_executable(reaction_wheel_driver_node src/reaction_wheel_driver_node.cpp)
target_compile_features(reaction_wheel_driver_node PRIVATE cxx_std_17)
target_compile_options(reaction_wheel_driver_node PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(reaction_wheel_driver_node PRIVATE include)
target_link_libraries(reaction_wheel_driver_node fss_hal::fss_hal_cpp nlohmann_json::nlohmann_json)
ament_target_dependencies(reaction_wheel_driver_node rclcpp ament_index_cpp std_msgs std_srvs)

install(TARGETS valve_driver_node reaction_wheel_driver_node DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(cpp_actuators_test ../../../tests/cpp_actuators_test.cpp)
  target_include_directories(cpp_actuators_test PRIVATE include)
  target_link_libraries(cpp_actuators_test fss_valve_map fss_hal::fss_hal_cpp)
  ament_add_gtest(cpp_plc_test ../../../tests/cpp_plc_test.cpp)
  target_link_libraries(cpp_plc_test fss_hal::fss_hal_cpp)
endif()
ament_package()
